<script>

  const inputs = document.querySelectorAll("select");


  inputs.forEach(input => input.addEventListener("change", calculate));


  function calculate() {

    const value = +document.getElementById("value").value;

    const wait = +document.getElementById("wait").value;

    const quality = +document.getElementById("quality").value;

    const experience = +document.getElementById("experience").value;

    const comeback = document.getElementById("return").value;


    let score =

      value * 0.4 +

      wait * 0.25 +

      quality * 0.25 +

      experience * 0.1;


    let stars = Math.round(score * 2) / 2;


    if (comeback === "no" && stars > 3) stars = 3;


    document.getElementById("result").textContent =

      `Final Score: ${stars} ⭐`;

  }
